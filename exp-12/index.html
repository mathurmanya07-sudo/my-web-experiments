mern-app/ client/ (React frontend) server/ (Express + MongoDB backend) # ---------- server/server.js ---------- const express = require("express"); const mongoose = require("mongoose"); const cors = require("cors"); const path = require("path"); require("dotenv").config(); const app = express(); app.use(cors()); app.use(express.json()); mongoose.connect(process.env.MONGO_URI) .then(() => console.log("MongoDB Atlas connected")); const Task = mongoose.model("Task", { title: String, done: Boolean }); app.get("/api/tasks", async (req, res) => res.json(await Task.find())); app.post("/api/tasks", async (req, res) => res.status(201).json(await Task.create(req.body))); // serve React build in production app.use(express.static(path.join(__dirname, "../client/build"))); app.get("*", (req, res) => res.sendFile(path.join(__dirname, "../client/build/index.html"))); app.listen(process.env.PORT || 5000); # ---------- Deployment steps (Heroku) ---------- cd client && npm run build && cd .. heroku create mern-task-app heroku config:set MONGO_URI="" git push heroku main heroku logs --tail heroku open # ---------- Deployment steps (AWS EC2) ---------- ssh -i key.pem ubuntu@ git clone && cd mern-app/server && npm install sudo npm install -g pm2 pm2 start server.js --name mern-app pm2 save && pm2 startup
